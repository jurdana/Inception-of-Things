1. Preparation (une seule fois)
-------------------------------

make prepare                               # 2e lancement : doit tout sauter ("already")

2. P1
-----
time make p1
# dans les logs : "[+] Using pre-downloaded K3s ..." sur les 2 VMs
# a la fin : 2 nodes Ready (pjurdanas control-plane, pjurdanasw)

cd p1
vagrant ssh pjurdanaS
    hostname                                               # pjurdanaS
    ip a show $(ip route | grep default | awk '{print $5}') # commande de la correction : doit montrer 192.168.56.110
    ip route                                               # une seule route default, via 192.168.56.1
    kubectl get nodes -o wide                              # 2 nodes Ready, INTERNAL-IP .110 / .111
    exit
vagrant ssh pjurdanaSW
    hostname                                               # pjurdanaSW
    ip a show $(ip route | grep default | awk '{print $5}') # 192.168.56.111
    systemctl status k3s-agent --no-pager                  # active (running)
    exit
cd ..
# lancer le testeur ici

make p1-clean

4. Si ca casse
--------------
# Pas d'internet dans la VM ("no internet through 192.168.56.1") :
make net
sudo nft list table ip iot_nat
sudo iptables -L DOCKER-USER -n            # si Docker est installe : 2 regles ACCEPT 192.168.56.0/24

# K3s ne demarre pas (dans la VM) :
sudo journalctl -u k3s --no-pager | tail -50       # (k3s-agent sur le worker)
sudo ausearch -m avc -ts recent                    # refus SELinux
ls -Z /usr/local/bin/k3s                           # doit etre container_runtime_exec_t

# Comparer sans le cache (installation en ligne) :
mv ~/.cache/iot/k3s ~/.cache/iot/k3s.off && make p1 ; mv ~/.cache/iot/k3s.off ~/.cache/iot/k3s
